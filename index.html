const SHEET_NAME = "Hoja 1"; // Asegúrate de que coincida con el nombre de tu pestaña function doGet(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const data = sheet.getDataRange().getValues(); if (data.length <= 1) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } const headers = data[0]; const rows = data.slice(1); const result = rows.map(row => { let item = {}; headers.forEach((header, i) => { let val = row[i]; if (val === "") val = null; item[header] = val; }); return item; }); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const body = JSON.parse(e.postData.contents); const headers = [ "nombre", "cantidad", "formato", "unidad", "lider", "fLider", "linkLider", "jumbo", "fJumbo", "linkJumbo", "mayorista", "fMayorista", "linkMayorista" ]; // Si la hoja está totalmente vacía, crea los encabezados if (sheet.getLastRow() === 0) { sheet.appendRow(headers); } if (body.action === "saveAll") { // Reemplaza toda la lista con el arreglo nuevo const lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.deleteRows(2, lastRow - 1); } if (body.productos && body.productos.length > 0) { const rows = body.productos.map(p => headers.map(h => p[h] !== undefined && p[h] !== null ? p[h] : "")); sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } if (body.action === "clear") { const lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.deleteRows(2, lastRow - 1); } return ContentService.createTextOutput(JSON.stringify({ status: "cleared" })) .setMimeType(ContentService.MimeType.JSON); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
